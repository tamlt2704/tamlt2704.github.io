              <h2 className="text-sm font-semibold text-gray-500 uppercase">
                {series.replace(/-/g, " ")}

              <button onClick={() => setOpen(false)} className="text-xl text-gray-400">

              </button>
            </div>

                const slug = ch.replace(".md", "");

                  <Link
                    key={ch}

                  </Link>
                );
              })}
            </nav>
          </aside>
        </div>

    </>
  );
