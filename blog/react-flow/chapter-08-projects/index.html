Workflow Builder (n8n/Zapier Style)

import { useCallback, useState, DragEvent } from "react";
import {
  ReactFlow,
  Background,
  Controls,
  Handle,
  Position,
  addEdge,
  applyNodeChanges,
  applyEdgeChanges,
  ReactFlowProvider,
  useReactFlow,
} from "@xyflow/react";
import type {
  Node,
  Edge,
  NodeProps,
  NodeTypes,
  OnNodesChange,
  OnEdgesChange,
  OnConnect,
} from "@xyflow/react";
import "@xyflow/react/dist/style.css";

function TriggerNode({ data }: NodeProps) {
  return (
    <div className="rounded-lg border-2 border-amber-500 bg-amber-100 px-4 py-2">
      <span className="text-xs font-bold text-amber-700">TRIGGER</span>
      <p className="text-sm">{data.label}</p>
      <Handle type="source" position={Position.Bottom} />
    </div>
  );
}

function ActionNode({ data }: NodeProps) {
  return (
    <div className="rounded-lg border-2 border-blue-400 bg-blue-50 px-4 py-2">
      <Handle type="target" position={Position.Top} />
      <span className="text-xs font-bold text-blue-600">ACTION</span>
      <p className="text-sm">{data.label}</p>
      <Handle type="source" position={Position.Bottom} />
    </div>
  );
}

const nodeTypes: NodeTypes = { trigger: TriggerNode, action: ActionNode };

let id = 0;
const getId = () => `node_${id++}`;

function Sidebar() {
  const onDragStart = (e: DragEvent, type: string, label: string) => {
    e.dataTransfer.setData("type", type);
    e.dataTransfer.setData("label", label);
  };
  return (
    <aside className="w-52 space-y-2 border-r bg-gray-50 p-4">
      <p className="text-sm font-bold">Nodes</p>
      <div
        className="cursor-grab rounded bg-amber-100 p-2 text-sm"
        draggable
        onDragStart={(e) => onDragStart(e, "trigger", "Webhook")}
      >
        Webhook Trigger
      </div>
      <div
        className="cursor-grab rounded bg-blue-50 p-2 text-sm"
        draggable
        onDragStart={(e) => onDragStart(e, "action", "HTTP Request")}
      >
        HTTP Request
      </div>
      <div
        className="cursor-grab rounded bg-blue-50 p-2 text-sm"
        draggable
        onDragStart={(e) => onDragStart(e, "action", "Send Email")}
      >
        Send Email
      </div>
    </aside>
  );
}

function Flow() {
  const [nodes, setNodes] = useState<Node[]>([]);
  const [edges, setEdges] = useState<Edge[]>([]);
  const { screenToFlowPosition } = useReactFlow();

  const onNodesChange: OnNodesChange = useCallback(
    (c) => setNodes((n) => applyNodeChanges(c, n)),
    [],
  );
  const onEdgesChange: OnEdgesChange = useCallback(
    (c) => setEdges((e) => applyEdgeChanges(c, e)),
    [],
  );
  const onConnect: OnConnect = useCallback((c) => setEdges((e) => addEdge(c, e)), []);

  const onDrop = useCallback(
    (event: DragEvent) => {
      event.preventDefault();
      const type = event.dataTransfer.getData("type");
      const label = event.dataTransfer.getData("label");
      const position = screenToFlowPosition({ x: event.clientX, y: event.clientY });
      setNodes((nds) => [...nds, { id: getId(), type, position, data: { label } }]);
    },
    [screenToFlowPosition],
  );

  return (
    <div className="h-full flex-1">
      <ReactFlow
        nodes={nodes}
        edges={edges}
        nodeTypes={nodeTypes}
        onNodesChange={onNodesChange}
        onEdgesChange={onEdgesChange}
        onConnect={onConnect}
        onDrop={onDrop}
        onDragOver={(e) => e.preventDefault()}
        fitView
      >
        <Background />
        <Controls />
      </ReactFlow>
    </div>
  );
}

export default function App() {
  return (
    <ReactFlowProvider>
      <div className="flex h-screen w-screen">
        <Sidebar />
        <Flow />
      </div>
    </ReactFlowProvider>
  );
}

Org Chart with Expand/Collapse

import { useState, useCallback } from "react";
import { ReactFlow, Background, Handle, Position, applyNodeChanges } from "@xyflow/react";
import type { Node, Edge, NodeProps, NodeTypes, OnNodesChange } from "@xyflow/react";
import "@xyflow/react/dist/style.css";

type OrgData = { label: string; role: string; children?: string[] };

function OrgNode({ data, id }: NodeProps<Node<OrgData>>) {
  const [expanded, setExpanded] = useState(true);
  return (
    <div className="min-w-[120px] rounded-lg border-2 border-slate-300 bg-white px-4 py-2 text-center">
      <Handle type="target" position={Position.Top} />
      <p className="text-sm font-bold">{data.label}</p>
      <p className="text-xs text-gray-500">{data.role}</p>
      {data.children && data.children.length > 0 && (
        <button onClick={() => setExpanded(!expanded)} className="mt-1 text-xs text-blue-500">
          {expanded ? "▼ Collapse" : "▶ Expand"}
        </button>
      )}
      <Handle type="source" position={Position.Bottom} />
    </div>
  );
}

const nodeTypes: NodeTypes = { org: OrgNode };

const nodes: Node[] = [
